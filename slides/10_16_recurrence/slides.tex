\vfill 

\only<3>{\colorbox{green!30}{\textbf{Solution.}} 
From the last slide, we have
%
\begin{align*}
m_k &= m_{k-1} + 1 + m_{k-1} \\
	&= 2 m_{k-1} + 1
\end{align*}
}

\only<4->{\colorbox{yellow!30}{\textbf{Poll.}} We have derived a recursive formula for $m_k$ as \[m_k = 2 m_{k-1} + 1. \] 

So long would it take to solve the puzzle? 
}

\only<5->{\colorbox{green!30}{\textbf{Solution.}} Using the techniques from the textbook, we can derive the explicit formula
\[ m_k = 2^k - 1.\]}

\only<8->{Using this, we find  $m_{64} \approx 1.88 \times 10^{19}$.  In other words, if you work rapidly, and move one disk every second, the total number of time it would take to solve the puzzle with 64 disks is: \alert{584.5 billion years}! }

\begin{frame}[fragile]{Recursive programs are slow}
\footnotesize 
Consider the program to solve  $a_n = 3 a_{n-1} - 2 a_{n-2}$ with initial conditions $a_0=1, a_1=5$:
\begin{python}
def get_term(n):
    if (n < 0):
    	print("Illegal argument")
    elif (n==0):
    	return 1
    elif (n==1):
    	return 5
    else:
        return 3*get_term(n-1) - 2*get_term(n-2)
\end{python}
	
\pause \colorbox{yellow!30}{\textbf{Poll.}} How many times is this program called in order to calculate $a_n$?
\pause \vfill 

\colorbox{green!30}{\textbf{Solution.}} 
From the last line, the program is called
\[ b_n = b_{n-1} + b_{n-2} + 1\]
times to get term $a_n$.  The first few terms are 1,1,3,5,9,15,25.\\

\vfill \pause 
\vspace{.2cm}
\colorbox{red!30}{\textbf{Problem.}} This program is expensive to run! To get the 50-th term $a_{50}$, the program must be called $b_{50} \approx 40.7$ billion times!
 
\end{frame}

\begin{frame}{Most important results from the textbook -- Part 1}

\begin{mygreenbox}[title=\text{Solving first-order recurrences (Props 23.1, 23.3)}]
All solutions to recurrence relation $a_n = s a_{n-1} + t$ have the form
\[
a_n = 
\begin{cases}
c_1 s^n + c_2, & \text{ if } s \neq 1 \\
a_0 + nt, & \text{ if } s = 1 \\
\end{cases}
\]
\end{mygreenbox}
\vfill 
\pause 

\only<2->{
\colorbox{blue!30}{\textbf{Remark.}} In the first group exercise, we'll apply this to solve the Tower of Hanoi problem!
}

\only<3->{
\begin{mygreenbox}[title=\text{Solving second-order recurrences (Thms 23.5, 23.9)}]
All solutions to recurrence relation $a_n = s_1 a_{n-1} + s_2 a_{n-2}$ have the form
\[
a_n = 
\begin{cases}
c_1 r_1^n + c_2r_2^n, & \text{ if } r_1 \neq r_2 \\
c_1 r^n + c_2 n r^n, & \text{ if } r_1 =  r_2 \defeq r \\
\end{cases}
\]
%
where $r_1, r_2$ are roots of $x^2 - s_1x - s_2 =0$.
\end{mygreenbox}
}

\end{frame}

\begin{frame}{Solution to group exercise \#1}
\footnotesize 
 
\colorbox{red!30}{\textbf{Problem.}} Solve the Tower of Hanoi recursion.  
\vfill 

\colorbox{blue!30}{\textbf{Remark.}} Recall that the Tower of Hanoi recursion is 
\begin{align}
m_k = 2 m_{k-1} + 1,
\label{eq:hanoi_recursion}	
\end{align}

where $m_k$ is the minimum number of moves needed to transfer a tower of $k$ disks from one pole to another. 
\vfill 

\colorbox{green!30}{\textbf{Solution.}} By Prop 23.1, we know the solution satisfies
\begin{align}
 m_k = c_1 2^k + c_2
 \label{eq:hanoi_solution}	
\end{align}

for some constants $c_1, c_2$.  From the structure of the Tower of Hanoi problem, we have the initial condition $m_1=1$. Propagating this through the recursion in \Eqref{eq:hanoi_recursion}, we obtain $m_2=3$.  By substitution into the form of the solution in \Eqref{eq:hanoi_solution}, we obtain
%
\begin{align*}
1 &= 2c_1 + c_2 \\
3 &= 4c_1 + c_2	
\end{align*}
%
Solving this system of equations yields $c_1=1, c_2=-1$, and so
\[m_k = 2^k - 1.\]

\end{frame}



\begin{frame}{Solution to group exercise \#2: Overview}
\footnotesize 
Group exercise \#1 asks us to deduce explicit formulas for $a_n$ from recurrence relations.  

\vfill 

\colorbox{green!30}{\textbf{First Order Recurrence.}} A first order recurrence takes the form
\[ a_n = s a_{n-1}+ t\]
The explicit solution depends on the context 
\[
\begin{array}{cccccc}
\textbf{Context} && \textbf{Solution} && \textbf{Justification} \\
\text{If } s \neq 1 & & a_n = c_1 s^n + c_2 && \text{Prop 23.1} \\
\text{If } s =1 & & a_n = a_0 + nt && \text{Prop 23.3} 
\end{array}
\]

\vfill 

\colorbox{green!30}{\textbf{Second Order Recurrence.}}  A second order recurrence takes the form
\[ a_n = s_1 a_{n-1} + s_2 a_{n-2} \]
The explicit solution depends on the context. First, we solve the quadratic formula $x^2 - s_1 x - s_2 =0$ to find roots $r_1$ and $r_2$.
\[
\begin{array}{cccccc}
\textbf{Context} && \textbf{Solution} && \textbf{Justification} \\
\text{If } r_1 \neq r_2  & & a_n = c_1 r_1^n + c_2 r_2^n && \text{Thm 23.5} \\
\text{If } r_1 = r_2 \defeq r & & a_n = c_1 r^n + c_2 n r^n && \text{Thm 23.9} 
\end{array}
\]

\vfill 

\colorbox{red!30}{\textbf{Remark.}} Whenever the solution involves unknown constants (denoted $c_1$ and $c_2$), you must solve for them. You do this by plugging in the first two values for the sequence ($a_0$ and $a_1$) along with other known values (some subset of $\set{s^n, t, r_1^n, r_2^n, r^n, nr^n}$) to form a system of two equations with two unknowns.  Then solve the system (e.g. using substitution).   

\end{frame}


\begin{frame}{Solution to group exercise \#2a}

\colorbox{red!30}{\textbf{Problem.}} Solve the recurrence relation $a_n = 3 a_{n-1} -1$ with initial condition $a_0=10$ to give an explicit formula for $a_n$. 

\vfill 
	
\colorbox{green!30}{\textbf{Solution.}} By Prop 23.1, the solution takes the form 
%
\begin{align*}
a_n &= c_1 s^n + c_2 \\
\implies a_n &= c_1 \cdot 3^n + c_2
\end{align*}
%
Substituting in the provided initial condition $a_0=10$ and the next term of the sequence $a_1 =29$, we obtain the system of equations
%
\begin{align*}
10 &= c_1 + c_2 \\
29 &= 3c_1 + c_2
\end{align*}
%
Solving this system yields $c_1 = 9.5$ and $c_2 = 0.5$.  Hence, the final solution is
\[ a_n = 9.5 \cdot 3^n + 0.5 \]


\end{frame}

\begin{frame}{Solution to group exercise \#2c}
\footnotesize
\colorbox{red!30}{\textbf{Problem.}} Solve the recurrence relation $a_n = 8 a_{n-1} - 15a_{n-2}$  with initial conditions $a_0=1$ and $a_1=4$ to give an explicit formula for $a_n$. 
\vfill  \vfill 
\colorbox{green!30}{\textbf{Solution.}} First, we solve the quadratic formula 
%
\begin{align*}
& x^2 - s_1 x - s_2 =0  \\
\implies & x^2 - 8 x + 15 =0  \\
\implies & (x-5)(x-3) =0  
\end{align*}
%
Hence the roots are $r_1=5$ and $r_2=3$.  Since these roots are distinct, we apply Thm 23.5, which tell us that the solution takes the form 
%
\begin{align*}
a_n &= c_1 r_1^n + c_2 r_2^n \\
\implies a_n &= c_1 5^n + c_2 3^n 
\end{align*}
%
Substituting in the provided initial conditions  $a_0=1$ and $a_1=4$, we obtain the system of equations
%
\begin{align*}
1 &= c_1 + c_2 \\
4 &= 5c_1 + 3c_2
\end{align*}
%
Solving this system yields $c_1 = \half$ and $c_2 = \half$.  Hence, the final solution is
\[ a_n = \half \cdot  5^n + \half \cdot 3^n  \]


\end{frame}

\begin{frame}{Solution to group exercise \#2d}
\footnotesize
\colorbox{red!30}{\textbf{Problem.}} Solve the recurrence relation $a_n = 4 a_{n-1} -4 a_{n-2}$  with initial conditions $a_0=5$ and $a_1=1$ to give an explicit formula for $a_n$. 
\vfill  \vfill 
\colorbox{green!30}{\textbf{Solution.}} First, we solve the quadratic formula 
%
\begin{align*}
& x^2 - s_1 x - s_2 =0  \\
\implies & x^2 - 4 x + 4 =0  \\
\implies & (x-2)(x-2) =0  
\end{align*}
%
Hence the roots are $r_1=2$ and $r_2=2$.  Since these roots are identical, we apply Thm 23.9, which tell us that the solution takes the form 
%
\begin{align*}
a_n &= c_1 r^n + c_2 n r^n \\
\implies a_n &= c_1 2^n + c_2 n 2^n 
\end{align*}
%
Substituting in the provided initial conditions  $a_0=5$ and $a_1=1$, we obtain the equations
%
\begin{align*}
5 &= c_1  \\
1 &= 2c_1 + 2c_2
\end{align*}

\begin{frame}{Solution to group exercise \#3}
\small 
\colorbox{red!30}{\textbf{Problem.}} [...] Show how to \enquote{outsmart} the intelligence test by finding a polynomial expression (of degree 3) for $a_n$ such that $a_0=1, a_1=2$, $a_2=4, a_3=8$, but $a_4=15$. \\
\vfill 
\colorbox{green!30}{\textbf{Solution.}} We apply Scheinerman Theorem 23.17.
\vspace{-.25cm}
\[
\begin{array}{ccccccccccc}
a:& & 1 &     & 2 &     & 4 &     & 8 &  &15 \\
\Delta a:& &  & 1    &  & 2    &  &  4   &  & 7  & \\
\Delta^2 a:& &  &     & 1 &     & 2 &     &  3 &   & \\
\Delta^3 a:& &  &     &  &  1   & &   1  &   &   & \\
\Delta^4 a:& &  &     &  &     & 0 &     &   &   & \\
\end{array}
\]
Hence $k=3$ and 
%
\begin{align*}
a_n &= a_0 \binom{n}{0} +	\Delta a_0 \binom{n}{1} + 		\Delta^2 a_0 \binom{n}{2} + 		\Delta^3 a_0 \binom{n}{3} \\
&= 1 \binom{n}{0} +	1  \binom{n}{1} + 		1 \binom{n}{2} + 		1 \binom{n}{3} && \scripttext{(Substitute coefs from table)} \\
&= \big[1 \cdot 1\big] +	 \big[1 \cdot n\big]  + 		\big[1 \cdot \frac{n (n-1)}{2}\big] + 		\big[1 \cdot \frac{n (n-1) (n-2)}{6}\big] && \scripttext{(Expand binomial coefs.)} \\
&= \frac{1}{6} n^3 + \frac{5n}{6} + 1 && \scripttext{(Algebra)}
\end{align*}


\end{frame}
